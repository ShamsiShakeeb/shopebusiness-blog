<!-- JavaScript for Language Toggle Logic -->
<script>
    function switchLanguage(lang) {
        // Get the content wrappers
        const contentEn = document.getElementById('content-en');
        const contentBn = document.getElementById('content-bn');
        
        // Get the buttons
        const btnEn = document.getElementById('btn-en');
        const btnBn = document.getElementById('btn-bn');

        if (lang === 'bn') {
            // Show Bengali, Hide English
            contentEn.style.display = 'none';
            contentBn.style.display = 'block';
            
            // Update button styles
            btnEn.classList.remove('active');
            btnBn.classList.add('active');
        } else {
            // Show English, Hide Bengali
            contentEn.style.display = 'block';
            contentBn.style.display = 'none';
            
            // Update button styles
            btnBn.classList.remove('active');
            btnEn.classList.add('active');
        }
    }
</script>
